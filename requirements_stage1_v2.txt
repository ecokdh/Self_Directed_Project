# requirements_stage1.txt — 1단계(1~6주차), 노트북 CPU 환경
#
# 검증 환경: Windows 11(한글) · Python 3.11 · GPU 없음
#            모델 다운로드와 추론까지 실행 확인 (3.12 · 3.13 에서도 설치 가능)
#
# ★ Python 3.14 는 쓸 수 없습니다 — granite-tsfm 0.3.8 이 `<3.14` 를 선언하여
#   6주차 TTM 이 설치되지 않습니다. 3.11 을 쓰십시오.
#
# 설치:
#   python -m venv .venv
#   Windows:      .venv\Scripts\activate
#   macOS/Linux:  source .venv/bin/activate
#   pip install -r requirements_stage1.txt
#
# Linux 사용자만: torch 기본 휠이 CUDA 번들(수 GB)이므로 다음을 붙이십시오.
#   pip install -r requirements_stage1.txt --index-url https://download.pytorch.org/whl/cpu
# Windows·macOS 는 PyPI 기본이 CPU 빌드라 추가 조치가 필요 없습니다.
#
# 설치 후 확인 — None 이 나와야 CPU 빌드입니다.
#   python -c "import torch; print(torch.__version__, torch.version.cuda)"
#
# ★ 버전을 임의로 올리지 마십시오. 특히 transformers 는 아래 사유로 상한이 걸려 있습니다.

pandas==2.3.3
numpy==2.3.5
pyarrow==22.0.0
scikit-learn==1.7.2
lightgbm==4.7.0
shap==0.51.0                 # ★ 0.52.0 은 Python >=3.12 요구. 3.11 에서는 설치 불가
matplotlib==3.11.0
jupyterlab==4.5.6

# ★ 상한 필수 — granite-tsfm 0.3.8 이 `torch>=2.10,<2.11` 을 선언합니다.
#   2.11 이상을 쓰면 6주차 TTM 과 의존성이 충돌해 설치 자체가 실패합니다.
torch==2.10.0

# ★ 상한 필수 — granite-tsfm 0.3.8 의 선언은 `transformers[torch]>=4.57.6,<6` 뿐이라
#   5.x 가 그대로 허용됩니다. 핀이 없으면 설치 시점의 최신 transformers 가 딸려 들어오고,
#   조합에 따라 6주차 TTM 이 로드 단계에서 죽습니다.
#   (재현 사례: granite-tsfm 0.2.23 + transformers 5.x
#    → AttributeError: ... has no attribute 'all_tied_weights_keys')
transformers==4.57.6

chronos-forecasting==2.3.1   # 5주차 Chronos-2. 없으면 5주차 전체가 import 에서 멈춤
granite-tsfm==0.3.8          # 6주차 TTM. 없으면 과제 6-1 실행 불가
psutil==7.1.3                # 과제 6-2 최대 RAM 측정
